// Kode Apps Script Backend untuk Google Sheets function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet(); // Lookup Pegawai if (e.parameter.nip) { var empSheet = sheet.getSheetByName("Pegawai"); if (!empSheet) return jsonResp({status: "error", message: "Sheet Pegawai tidak ditemukan"}); var data = empSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][0].toString() == e.parameter.nip) { return jsonResp({status: "found", nama: data[i][1], nik: data[i][2]}); } } return jsonResp({status: "not_found"}); } // Load SK Data if (e.parameter.sheet == "admin_data") { var skSheet = sheet.getSheetByName("Pengumpulan_SK"); if (!skSheet) return jsonResp([]); var data = skSheet.getDataRange().getValues(); var list = []; for (var j = 1; j < data.length; j++) { list.push({ row: j + 1, nip: data[j][0], nik: data[j][1], nama: data[j][2], ruangan: data[j][3], nomor_sk: data[j][4], link_sk: data[j][5], status: data[j][6] || 'Pending' }); } return jsonResp(list); } // Update Status if (e.parameter.sheet == "update_status") { var skSheet = sheet.getSheetByName("Pengumpulan_SK"); var row = parseInt(e.parameter.row); var status = e.parameter.status; skSheet.getRange(row, 7).setValue(status); return jsonResp({status: "success"}); } } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Pengumpulan_SK"); if (!sheet) { sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("Pengumpulan_SK"); sheet.appendRow(["NIP", "NIK", "Nama", "Ruangan", "Nomor SK", "Link SK", "Status"]); } // Save File to Drive var folder = DriveApp.getFoldersByName("Berkas_SK_Pegawai"); var targetFolder = folder.hasNext() ? folder.next() : DriveApp.createFolder("Berkas_SK_Pegawai"); var blob = Utilities.newBlob(Utilities.base64Decode(data.fileData), data.fileMimeType, "SK_" + data.nip + "_" + data.nomor_sk + ".pdf"); var file = targetFolder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); sheet.appendRow([data.nip, data.nik, data.nama, data.ruangan, data.nomor_sk, file.getUrl(), "Pending"]); return jsonResp({status: "success", message: "Berkas SK berhasil dikirim!"}); } catch (err) { return jsonResp({status: "error", message: err.toString()}); } } function jsonResp(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); }
